\def\texnim{%
	\begingroup%
	\catcode32=9\relax%
	\catcode13=9\relax%
	\@ifnextchar[{\texnim@option}{\@ifnextchar({\@texnim}{\texnim@mode}}%
}
\def\texnim@title{%
	\texnim@ifundefined{texnim@title@flag}
		{\begin{center}\huge Nim on \TeX\end{center}\smallskip\smallskip}
		{\par\noindent}%
}
\long\def\texnim@option[#1]{%
	\texnim@ifmt{#1}
		{\@texnim}
		{\@texnim@option#1,\texnim@relax%
			\@ifnextchar({\@texnim}{\texnim@mode}}%
}
\def\@texnim@option#1,#2\texnim@relax{%
	\texnim@ifmt{#2}
		{\texnim@ifmt{#1}
			{\texnim@optionerror{2}}
			{\@texnim@option@a#1=\texnim@relax}}
		{\texnim@ifmt{#1}
			{\texnim@optionerror{2}}
			{\@texnim@option@a#1=\texnim@relax
				\texnim@ifundefined{texnim@option@error@flag}
					{\@texnim@option#2\texnim@relax}
					{}}}%
}
\def\@texnim@option@a#1=#2\texnim@relax{%
	\texnim@ifmt{#2}
		{\texnim@optionerror{1}}
		{\texnim@ifmt{#1}
			{\texnim@optionerror{1}}
			{\@texnim@option@b{#1}#2\texnim@relax}}%
}
\def\@texnim@option@b#1#2=#3\texnim@relax{%
	\texnim@ifmt{#3}
		{\texnim@ifmt{#2}
			{\texnim@optionerror{1}}
			{\texnim@ifundefined{texnim@option@\string#1@exists}
				{\texnim@optionerror{0}}
				{\texnim@ifundefined{texnim@\string#1@exists@flag}
					{\expandafter\def\csname texnim@\string#1@exists@flag\endcsname{#2}%
						\csname texnim@option@\string#1@exists\endcsname{#2}}
					{\texnim@optionerror{4}}}}}
		{\texnim@optionerror{3}}%
}
\def\texnim@mode#1{%
	\texnim@ifundefined{texnim@mode@\expandafter\@gobble\string#1@exists}
		{\texnim@modeerror}
		{\csname texnim@mode@\expandafter\@gobble\string#1@exists\endcsname}%
}
\long\def\@texnim(#1){%
	\catcode32=10\relax%
	\texnim@option@check%
	{\texnim@title%
		\texnim@ifundefined{texnim@history@A}
			{\texnim@check@error#1,\texnim@relax
				{\immediate\write\@auxout{^^J\@percentchar\space ---------- TeX Nim History ----------}%
					\immediate\write\@auxout{\string\gdef\string\texnim@turn{A}}%
					\texnim@ifundefined{texnim@consecutivewait@flag}
						{}
						{\immediate\write\@auxout{\string\gdef\string\texnim@consecutivewait{\texnim@consecutivewait@flag}}}%
					\texnim@ifundefined{texnim@currentwait}
						{}
						{\immediate\write\@auxout{\string\gdef\string\texnim@currentwait{\texnim@currentwait}}%
							\immediate\write\@auxout{\string\gdef\string\texnim@wait{\texnim@wait}}}%
					\texnim@next{#1}%
					\texnim@ifundefined{texnim@lastposition}
						{}
						{\immediate\write\@auxout{\unexpanded\expandafter\expandafter\expandafter{%
							\expandafter\expandafter\expandafter\gdef\expandafter\expandafter\expandafter\texnim@history@A\expandafter\expandafter\expandafter{%
							\expandafter\texnim@history@tex\expandafter{%
							\texnim@lastposition}{#1}}}}}%
					\immediate\write\@auxout{\@percentchar\space -------------------------------------^^J}}}
			{\texnim@check{#1}
				{\immediate\write\@auxout{^^J\@percentchar\space ---------- TeX Nim History ----------}%
					\expandafter\def\expandafter\texnim@turn\expandafter{\texnim@turn A}%
					\texnim@ifundefined{texnim@consecutivewait@flag}
						{}
						{\immediate\write\@auxout{\string\gdef\string\texnim@consecutivewait{\texnim@consecutivewait}}}%

						{}
						{\immediate\write\@auxout{\string\gdef\string\texnim@currentwait{\texnim@currentwait}}%
							\immediate\write\@auxout{\string\gdef\string\texnim@wait{\texnim@wait}}}%
					\immediate\write\@auxout{\string\gdef\string\texnim@turn{\texnim@turn}}%
					\texnim@history%
					$\hbox{You}\,\colon (\texnim@lastposition) \,\, \to~(#1)$\\%
					\expandafter\expandafter\expandafter\def\expandafter\expandafter\expandafter\texnim@history@temp\expandafter\expandafter\expandafter{\expandafter\texnim@history@you\expandafter{\texnim@lastposition}{(#1)}}%
					\texnim@next{#1}%
					\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter\def\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter\texnim@history@temp\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter{\expandafter\expandafter\expandafter\texnim@history@temp\expandafter\texnim@history@tex\expandafter{\texnim@lastposition}{#1}}%
					\immediate\write\@auxout{\unexpanded\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter{\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter\expandafter\gdef\expandafter\texnim@history@input\expandafter{\texnim@history@temp}{\csname texnim@history@\texnim@turn\endcsname}}}%
					\immediate\write\@auxout{\@percentchar\space -------------------------------------^^J}}
				{\immediate\write\@auxout{^^J\@percentchar\space ---------- TeX Nim History ----------}%
					\immediate\write\@auxout{\string\gdef\string\texnim@turn{\texnim@turn}}%
					\texnim@ifundefined{texnim@consecutivewait@flag}
						{}
						{\immediate\write\@auxout{\string\gdef\string\texnim@consecutivewait{\texnim@consecutivewait}}}%
					\texnim@ifundefined{texnim@currentwait}
						{}
						{\immediate\write\@auxout{\string\gdef\string\texnim@currentwait{\texnim@currentwait}}%

					\immediate\write\@auxout{\@percentchar\space -------------------------------------^^J}%

\def\texnim@option@maxwait@exists#1{%
	\def\texnim@maxwait@flag{#1}%
	\texnim@ifundefined{texnim@currentwait}
		{\ifcompare[0]<=[#1]
			{\def\texnim@wait{#1}\def\texnim@currentwait{#1}}

			{}
			{\def\texnim@maxwait@error@flag{Y}}}%
}
\def\texnim@maxwait@error{%

}

			{\edef\texnim@consecutivewait@flag{\string#1}}

				{\csname if\texnim@consecutivewait\endcsname \expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi%

					{\def\texnim@consecutivewait@error@flag{Y}}}
				{\csname if\texnim@consecutivewait\endcsname \expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi%
					{\def\texnim@consecutivewait@error@flag{Y}}
					{\def\texnim@consecutivewait@flag{false}}}}}%
}

\def\texnim@check#1{%
	\expandafter\texnim@check@a\expandafter\texnim@relax\texnim@lastposition,\texnim@relax#1,\texnim@relax%
}
\def\texnim@check@a#1\texnim@relax#2,#3\texnim@relax#4,#5\texnim@relax{%
	\texnim@ifmt{#3#5}
		{\texnim@ifmt{#4}
			{\def\texnim@errorvalue{1}\@secondoftwo}
			{\ifcompare[#4]>=[0]
				{\texnim@ifmt{#1}
					{\ifcompare[#2]>[#4]
						{\@firstoftwo}
						{\def\texnim@errorvalue{0}\@secondoftwo}}
					{\ifcompare[#2]=[#4]
						{\@firstoftwo}
						{\def\texnim@errorvalue{3}\@secondoftwo}}}
				{\def\texnim@errorvalue{2}\@secondoftwo}}}
		{\texnim@ifmt{#3}
			{\def\texnim@errorvalue{4}\@secondoftwo}
			{\texnim@ifmt{#5}
				{\def\texnim@errorvalue{4}\@secondoftwo}
				{\texnim@ifmt{#4}
					{\def\texnim@errorvalue{1}\@secondoftwo}
					{\ifcompare[#4]>=[0]
						{\texnim@ifmt{#1}
							{\ifcompare[#2]>=[#4]
								{\ifcompare[#2]=[#4]
									{\texnim@check@a\texnim@relax#3\texnim@relax#5\texnim@relax}
									{\texnim@check@a0\texnim@relax#3\texnim@relax#5\texnim@relax}}
								{\def\texnim@errorvalue{0}\@secondoftwo}}
							{\ifcompare[#2]=[#4]
								{\texnim@check@a0\texnim@relax#3\texnim@relax#5\texnim@relax}
								{\def\texnim@errorvalue{3}\@secondoftwo}}}
						{\def\texnim@errorvalue{2}\@secondoftwo}}}}}%
}
\def\texnim@check@terminal#1,#2\texnim@relax{%
	\texnim@ifmt{#2}
		{\texnim@ifmt{#1}
			{\@secondoftwo}
			{\ifcompare[0]=[#1]
				{\@firstoftwo}
				{\@secondoftwo}}}
		{\texnim@ifmt{#1}
			{\@secondoftwo}
			{\ifcompare[0]=[#1]
				{\texnim@check@terminal#2\texnim@relax}
				{\@secondoftwo}}}%
}
\def\texnim@check@error#1,#2\texnim@relax{%
	\texnim@ifmt{#2}
		{\texnim@ifmt{#1}
			{\texnim@errortext@numberless\@gobble}
			{\ifcompare[0]<=[#1]
				{}
				{\texnim@errortext@negativenumber\@gobble}}}
		{\texnim@ifmt{#1}
			{\texnim@errortext@numberless\@gobble}
			{\ifcompare[0]<=[#1]
				{\texnim@check@error#2\texnim@relax}
				{\texnim@errortext@negativenumber\@gobble}}}%
}

				{\expandafter\texnim@texwin@b\expandafter{\romannumeral-`?\nimsum{#1}{#3}}{#2}{#3}#4\texnim@relax}

}
\def\texnim@texwin@a#1#2#3{%
	$\hbox{\TeX}\colon (#2#3) \,\, \to~(#2#1)$\\\\%
	\texnim@check@terminal#2#1,\texnim@relax
		{$\hbox{\TeX}\colon\hbox{You lose.}$\\}
		{$\hbox{You}\,\colon (#2#1) \,\, \to~\hbox{?}$\\}%
	\def\texnim@lastposition{#2#1}%
	\immediate\write\@auxout{\unexpanded{\gdef\texnim@lastposition{#2#1}}}%
}
\def\texnim@texwin@b#1#2#3#4,\texnim@relax{%
	$\hbox{\TeX}\colon (#2#3,#4) \,\, \to~(#2#1,#4)$\\\\%
	\texnim@check@terminal#2#1,#4,\texnim@relax
		{$\hbox{\TeX}\colon\hbox{You lose.}$\\}
		{$\hbox{You}\,\colon (#2#1,#4) \,\, \to~\hbox{?}$\\}%
	\def\texnim@lastposition{#2#1,#4}%
	\immediate\write\@auxout{\unexpanded{\gdef\texnim@lastposition{#2#1,#4}}}%
}
